{%- comment -%} Jekyll formatea fechas solo en ingles. Se indexa un array propio con el numero de mes. Uso: {% include fecha.html date=page.date %} {%- endcomment -%} {%- assign meses = "enero,febrero,marzo,abril,mayo,junio,julio,agosto,septiembre,octubre,noviembre,diciembre" | split: "," -%} {%- assign m = include.date | date: "%-m" | minus: 1 -%} {{ include.date | date: "%-d" }} de {{ meses[m] }} de {{ include.date | date: "%Y" }}
